def read_data(path):
    with open(path, encoding="utf-8") as f:
        return [line.strip() for line in f]

def analyze(lines):
    cnt = {}
    for line in lines:
        for c in line:
            cnt[c] = cnt.get(c, 0) + 1
    return sorted(cnt.items(),
                  key=lambda x: x[1], reverse=True)

def write_result(path, data):
    with open(path, "w", encoding="utf-8") as f:
        for k, v in data:
            f.write(f"{k}: {v}\n")

lines = read_data("data.txt")
stats = analyze(lines)
write_result("result.txt", stats)
📌

📌 核心考点

  • lambda：匿名函数，lambda 参数列表: 表达式
  • 常用于 sorted() / map() / filter() 的 key 参数
  • 只能写单条表达式，不能用 if/while/return
  • 闭包陷阱：lambda 引用循环变量
📂 lambda 实战复制
# 排序：按字典值从大到小
sorted(cnt.items(), key=lambda x: x[1], reverse=True)

# map：每个元素乘 2
list(map(lambda x: x*2, [1,2,3]))                # [2,4,6]

# filter：保留奇数
list(filter(lambda x: x % 2 == 1, range(10)))       # [1,3,5,7,9]

# 闭包陷阱：循环变量
funcs = [lambda i=i: i for i in range(3)]
print([f() for f in funcs])                  # [0, 1, 2]（用 i=i 快照）
🚨
易错：funcs = [lambda: i for i in range(3)] 不加 i=i，所有 lambda 都返回最后那个 2。
